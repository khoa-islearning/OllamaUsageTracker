OllamaUsageRemainingBalanceWidget/OllamaUsageBalance.swift · 1–24 of 173 lines · 24 shown

import WidgetKit
import SwiftUI
import Security

// MARK: - Keychain

enum Keychain {
    static let service = "ollama-api"
    static let account = "widget"

    static func readKey() -> String? {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: service,
            kSecAttrAccount as String: account,
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne
        ]
        var out: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &out)
        guard status == errSecSuccess, let d = out as? Data else { return nil }
        return String(data: d, encoding: .utf8)
    }
}
SIDE NOTES

Now the product core, top-down. Imports map 1:1 to capability.

  • WidgetKit = the widget SDK · SwiftUI = templating · Security = the Keychain C API.
  • service "ollama-api" + account "widget" — the exact coordinates the setup command and this read share. One character off ⇒ errSecItemNotFound.
  • readKey(): kSecReturnData = true — the only line in the widget that can unlock/prompt the Keychain (first run ⇒ the "Always Allow" dialog).
  • kSecMatchLimitOne: exact-item lookup, no scan.
  • Gotcha: the widget never queries attributes — presence probing lives in SetupPanel (Stop 13) so no prompt can ever fire at launch.

// MARK: - API model (matches ollama.com/api/balance exactly)

struct BalanceResponse: Codable {
    let included: Included
    let purchased: Purchased

    struct Included: Codable {
        let balance_usd: Double
        let allowance_usd: Double
        let period: Period
    }
    struct Purchased: Codable {
        let balance_usd: Double
    }
    struct Period: Codable {
        let from: String
        let until: String
    }
}

struct BalanceEntry: TimelineEntry {
    let date: Date
    var remaining: Double = 0
    var allowance: Double = 0
    var purchased: Double = 0
    var renews: String = ""
    var error: String? = nil
}
SIDE NOTES

Read data before the code that produces it — then the provider reads like filling a form.

  • BalanceResponse mirrors the wire JSON 1:1 (snake_case) ⇒ synthesized Codable needs zero CodingKeys in this file.
  • Nesting = JSON nesting: included = granted allowance, purchased = top-up credit.
  • BalanceEntry is the layer boundary: a plain value struct with defaults (partial updates safe) and error: String? — errors are data because WidgetKit has no error channel.
  • renews is pre-formatted ⇒ the view never parses dates (unit-testable with no UI).
  • TS: a zod schema plus a reducer output; the mapper is one line — remaining = included + purchased.

OllamaUsageRemainingBalanceWidget/OllamaUsageBalance.swift · 54–97 of 173 lines · 44 shown

// MARK: - Provider

struct BalanceProvider: TimelineProvider {
    func placeholder(in context: Context) -> BalanceEntry {
        BalanceEntry(date: .now, remaining: 59.89, allowance: 60, renews: "Dec 4")
    }
    func getSnapshot(in context: Context, completion: @escaping (BalanceEntry) -> Void) {
        fetchBalance(completion)
    }
    func getTimeline(in context: Context, completion: @escaping (Timeline<BalanceEntry>) -> Void) {
        fetchBalance { entry in
            let next = Calendar.current.date(byAdding: .minute, value: 15, to: .now)!
            completion(Timeline(entries: [entry], policy: .after(next)))
        }
    }

    private func fetchBalance(_ done: @escaping (BalanceEntry) -> Void) {
        guard let key = Keychain.readKey(), !key.isEmpty,
              let url = URL(string: "https://ollama.com/api/balance") else {
            done(BalanceEntry(date: .now,
                error: "No API key in Keychain.\nRun:\nsecurity add-generic-password -s ollama-api -a widget -w"))
            return
        }
        var req = URLRequest(url: url)
        req.setValue("Bearer \(key)", forHTTPHeaderField: "Authorization")
        URLSession.shared.dataTask(with: req) { data, resp, _ in
            var entry = BalanceEntry(date: .now)
            if let http = resp as? HTTPURLResponse, http.statusCode == 401 {
                entry.error = "Invalid API key — rotate it and re-run the Keychain command"
            } else if let data,
                      let r = try? JSONDecoder().decode(BalanceResponse.self, from: data) {
                let until = ISO8601DateFormatter().date(from: r.included.period.until)
                entry.remaining = r.included.balance_usd + r.purchased.balance_usd
                entry.allowance = r.included.allowance_usd
                entry.purchased = r.purchased.balance_usd
                entry.renews = until?.formatted(.dateTime.month(.abbreviated).day()) ?? ""
            } else {
                entry.error = "Fetch failed"
            }
            DispatchQueue.main.async { done(entry) }
        }.resume()
    }
}
SIDE NOTES

The OS contract, three hooks: one line each in the callback chain.

  • placeholder → picker preview (the $59.89 demo entry) · getSnapshot → render-now · getTimeline → the scheduled refresh you can trace on the map.
  • policy: .after(next) — the OS stores it and wakes the extension later; nothing of yours runs between snapshots.
  • fetchBalance guard ladder: no key ⇒ instructive error entry · HTTP 401 ⇒ "rotate it" · decodable 200 ⇒ fields · anything else ⇒ "Fetch failed".
  • DispatchQueue.main.async: dataTask callbacks arrive off-main; the completion is marshaled back because view templating expects it.
  • Contract: exactly one completion call with ≥1 entry — this is why errors are entries, and why the timeline never goes blank.
  • Measured anchors (real log): 162 B request, 489 B response, ≈401 ms over HTTP/3.

    private func refreshKeychainPresence() {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: "ollama-api",
            kSecAttrAccount as String: "widget",
            kSecReturnData as String: false,
            kSecReturnAttributes as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne
        ]
        var result: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &result)

        switch status {
        case errSecSuccess:
            keychainPresence = .found
        case errSecItemNotFound:
            keychainPresence = .missing
        case errSecInteractionNotAllowed:
            keychainPresence = .locked
        default:
            keychainPresence = .error(status)
        }
    }

    private func startConnectionTest() {
        guard !isTesting else { return }
        isTesting = true
        testMessage = nil
        testSucceeded = false
        Task { await testConnection() }
    }
SIDE NOTES

The no-prompt probe, and the one-flight test starter.

  • refreshKeychainPresence: metadata-only query — kSecReturnData = false + attributes = true ⇒ keychaind never decrypts; no unlock UI is even possible from this call shape.
  • OSStatus mapping: success → found · itemNotFound → missing · interactionNotAllowed → locked ("exists but locked") · default → error(status).
  • startConnectionTest: single-flight guard, message reset, then Task { } keeps the async work on MainActor.
  • This probe is the launch-time behavior; the secret read is deliberately not here — see Stop 14.

    private func testConnection() async {
        defer { isTesting = false }

        let keyResult = readAPIKey()
        if keyResult.status == errSecItemNotFound {
            testMessage = "No API key yet — complete Step 1 first."
            return
        }
        if keyResult.status == errSecInteractionNotAllowed {
            testMessage = "Keychain is locked. Unlock it, then try again."
            return
        }
        guard keyResult.status == errSecSuccess else {
            testMessage = "Keychain access failed (error \(keyResult.status))."
            return
        }
        guard let key = keyResult.key, !key.isEmpty else {
            testMessage = "The Keychain item does not contain a readable API key."
            return
        }

        var request = URLRequest(url: URL(string: "https://ollama.com/api/balance")!)
        request.setValue("Bearer \(key)", forHTTPHeaderField: "Authorization")
        request.timeoutInterval = 20

        do {
            let (data, response) = try await URLSession.shared.data(for: request)
            guard let httpResponse = response as? HTTPURLResponse else {
                testMessage = "Ollama returned an invalid response."
                return
            }
            if httpResponse.statusCode == 401 {
                testMessage = "Invalid API key — rotate it (Step 1)."
                return
            }
            guard (200..<300).contains(httpResponse.statusCode) else {
                testMessage = "Ollama returned HTTP \(httpResponse.statusCode)."
                return
            }

            let balance = try JSONDecoder().decode(HostBalanceResponse.self, from: data)
            guard let renewalDate = parseRenewalDate(balance.included.period.until) else {
                testMessage = "Ollama returned an invalid renewal date."
                return
            }

            let remaining = balance.included.balanceUSD + balance.purchased.balanceUSD
            let renewal = renewalDate.formatted(.dateTime.month(.abbreviated).day())
            testMessage = "\(usd(remaining)) of \(usd(balance.included.allowanceUSD)) · renews \(renewal)"
            testSucceeded = true
        } catch let error as URLError {
            testMessage = "Network error: \(error.localizedDescription)"
        } catch is DecodingError {
            testMessage = "Ollama returned an unexpected balance response."
        } catch {
            testMessage = "Connection failed: \(error.localizedDescription)"
        }
    }

    private func readAPIKey() -> (key: String?, status: OSStatus) {
        let query: [String: Any] = [
            kSecClass as String: kSecClassGenericPassword,
            kSecAttrService as String: "ollama-api",
            kSecAttrAccount as String: "widget",
            kSecReturnData as String: true,
            kSecMatchLimit as String: kSecMatchLimitOne
        ]
        var result: CFTypeRef?
        let status = SecItemCopyMatching(query as CFDictionary, &result)
        guard status == errSecSuccess else { return (nil, status) }
        guard let data = result as? Data, let key = String(data: data, encoding: .utf8) else {
            return (nil, errSecSuccess)
        }
        return (key, errSecSuccess)
    }

    private func parseRenewalDate(_ value: String) -> Date? {
        let formatter = ISO8601DateFormatter()
        formatter.formatOptions = [.withInternetDateTime, .withFractionalSeconds]
        if let date = formatter.date(from: value) { return date }
        formatter.formatOptions = [.withInternetDateTime]
        return formatter.date(from: value)
    }

    private func usd(_ value: Double) -> String {
        value.formatted(.currency(code: "USD").precision(.fractionLength(2)))
    }
}

  • readAPIKey(): kSecReturnData = true — the one line that can consent-prompt; legitimate because the user clicked.
